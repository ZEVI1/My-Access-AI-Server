    32	        public void Connect(string databasePath)
    33	        {
    34	            if (!File.Exists(databasePath))
    35	                throw new FileNotFoundException($"Database file not found: {databasePath}");
    36
    37	            _currentDatabasePath = databasePath;
    38
    39	            try
    40	            {
    41	                // Use OleDb for connection to avoid COM interop dependency during connect
    42	                FileLogger.Log("AccessInteropService.Connect: opening OleDb connection");
    43	                var connectionString = $"Provider=Microsoft.ACE.OLEDB.12.0;Data Source={databasePath};";
    44	                _oleDbConnection = new OleDbConnection(connectionString);
    45	                _oleDbConnection.Open();
    46	                FileLogger.Log($"AccessInteropService.Connect: connected to '{databasePath}'.");
    47
    48	                // Do not require Access.Application during initial connection
    49	                _accessApplication = null;
    50	                _currentDatabase = null;
    51	                InvalidateSchemaCache();
    52	            }
    53	            catch (Exception ex)
    54	            {
    55	                // Clean up on failure
    56	                if (_oleDbConnection != null)
    57	                {
    58	                    try {
    59	                        _oleDbConnection.Close();
    60	                        _oleDbConnection.Dispose();
    61	                    }
    62	                    catch (Exception dbEx)
    63	                    {
    64	                        FileLogger.Log($"Error closing OleDb connection on failure: {dbEx.Message}");
    65	                    }
    66	                    _oleDbConnection = null;
    67	                }
    68
    69	                FileLogger.Log($"AccessInteropService.Connect: failed to connect to database: {ex.Message}");
    70	                throw new InvalidOperationException($"Failed to connect to database: {ex.Message}", ex);
    71	            }
    72	        }

    74	        public void Disconnect()
    75	        {
    76	            try
    77	            {
    78	                FileLogger.Log("AccessInteropService.Disconnect: starting cleanup.");
    79	                // Close OleDb connection first
    80	                if (_oleDbConnection != null)
    81	                {
    82	                    try
    83	                    {
    84	                        _oleDbConnection.Close();
    85	                        _oleDbConnection.Dispose();
    86	                    }
    87	                    catch (Exception ex)
    88	                    {
    89	                        FileLogger.Log($"Error closing OleDb connection: {ex.Message}");
    90	                    }
    91	                    _oleDbConnection = null;
    92	                }
    93
    94	                // Release COM objects in reverse order
    95	                // 1. Close the database
    96	                if (_currentDatabase != null)
    97	                {
    98	                    try
    99	                    {
   100	                        var databaseType = _currentDatabase.GetType();
   101	                        databaseType.InvokeMember("Close", BindingFlags.InvokeMethod, null, _currentDatabase, null);
   102	                    }
   103	                    catch (Exception ex)
   104	                    {
   105	                        FileLogger.Log($"Error closing database: {ex.Message}");
   106	                    }
   107	                    finally
   108	                    {
   109	                        ReleaseComObjectSafe(_currentDatabase);
   110	                        _currentDatabase = null;
   111	                    }
   112	                }
   113
   114	                // 2. Quit Access Application
   115	                if (_accessApplication != null)
   116	                {
   117	                    try
   118	                    {
   119	                        var accessType = _accessApplication.GetType();
   120
   121	                        try
   122	                        {
   123	                            var forms = accessType.InvokeMember("Forms", BindingFlags.GetProperty, null, _accessApplication, null);
   124	                            ReleaseComObjectSafe(forms);
   125	                        }
   126	                        catch { }
   127
   128	                        try
   129	                        {
   130	                            var reports = accessType.InvokeMember("Reports", BindingFlags.GetProperty, null, _accessApplication, null);
   131	                            ReleaseComObjectSafe(reports);
   132	                        }
   133	                        catch { }
   134
   135	                        accessType.InvokeMember("Quit", BindingFlags.InvokeMethod, null, _accessApplication, null);
   136	                    }
   137	                    catch (Exception ex)
   138	                    {
   139	                        FileLogger.Log($"Error quitting Access: {ex.Message}");
   140	                    }
   141	                    finally
   142	                    {
   143	                        ReleaseComObjectSafe(_accessApplication);
   144	                        _accessApplication = null;
   145	                    }
   146	                }
   147
   148	                _currentDatabasePath = null;
   149	                InvalidateSchemaCache();
   150	                FileLogger.Log("AccessInteropService.Disconnect: cleanup complete.");
   151	            }
   152	            catch (Exception ex)
   153	            {
   154	                FileLogger.Log($"Error during disconnect: {ex.Message}");
   155	            }
   156	        }

   185	        private static void ReleaseComObjectSafe(object? comObject)
   186	        {
   187	            if (comObject == null)
   188	                return;
   189
   190	            try
   191	            {
   192	                while (Marshal.ReleaseComObject(comObject) > 0) { }
   193	            }
   194	            catch { }
   195	        }

   198	        {
   199	            if (_accessApplication != null)
   200	                return _accessApplication;

   202	            if (string.IsNullOrEmpty(_currentDatabasePath))
   203	                return null;
   204
   205	            try
   206	            {

   223	        public List<TableInfo> GetTables()
   224	        {
   225	            if (!IsConnected) throw new InvalidOperationException("Not connected to database");
   226
   227	            lock (_schemaCacheLock)
   228	            {
   229	                if (_cachedTables != null)
   230	                    return _cachedTables;
   231	            }
   232
   233	            var tables = new List<TableInfo>();
   234
   235	            // Use OleDb to get table information
   236	            var schema = _oleDbConnection!.GetSchema("Tables");
   237
   238	            foreach (System.Data.DataRow row in schema.Rows)
   239	            {
   240	                var tableName = row["TABLE_NAME"].ToString();
   241	                if (!string.IsNullOrEmpty(tableName) && !tableName.StartsWith("~"))
   242	                {
   243	                    var fields = GetTableFields(tableName);
   244	                    tables.Add(new TableInfo
   245	                    {
   246	                        Name = tableName,
   247	                        Fields = fields,
   248	                        RecordCount = GetTableRecordCount(tableName)
   249	                    });
   250	                }
   251	            }
   252
   253	            lock (_schemaCacheLock)
   254	            {
   255	                _cachedTables = tables;
   256	            }
   257
   258	            return tables;
   259	        }

   261	        public List<QueryInfo> GetQueries()
   262	        {
   263	            if (!IsConnected) throw new InvalidOperationException("Not connected to database");
   264
   265	            lock (_schemaCacheLock)
   266	            {
   267	                if (_cachedQueries != null)
   268	                    return _cachedQueries;
   269	            }
   270
   271	            var queries = new List<QueryInfo>();
   272	            var schema = _oleDbConnection!.GetSchema("Views");
   273
   274	            foreach (System.Data.DataRow row in schema.Rows)
   275	            {
   276	                var queryName = row["TABLE_NAME"]?.ToString();
   277	                if (string.IsNullOrEmpty(queryName))
   278	                    continue;
   279
   280	                string sql = string.Empty;
   281	                dynamic? currentDb = null;
   282	                dynamic? queryDef = null;
   283	                try
   284	                {
   285	                    var accessApp = EnsureAccessApplication();
   286	                    if (accessApp != null)
   287	                    {
   288	                        currentDb = accessApp.CurrentDb();
   289	                        queryDef = currentDb.QueryDefs[queryName];
   290	                        if (queryDef != null)
   291	                        {
   292	                            sql = queryDef.SQL ?? string.Empty;
   293	                        }
   294	                    }
   295	                }
   296	                catch { }
   297	                finally
   298	                {
   299	                    ReleaseComObjectSafe(queryDef);
   300	                    ReleaseComObjectSafe(currentDb);
   301	                }
   302
   303	                queries.Add(new QueryInfo
   304	                {
   305	                    Name = queryName,
   306	                    SQL = sql,
   307	                    Type = "Query"
   308	                });
   309	            }
   310
   311	            lock (_schemaCacheLock)
   312	            {
   313	                _cachedQueries = queries;
   314	            }
   315
   316	            return queries;
   317	        }

   319	        public object ExecuteSql(string sql, List<object?>? parameters = null, string mode = "select")
   320	        {
   321	            if (!IsConnected) throw new InvalidOperationException("Not connected to database");
   322	            if (string.IsNullOrWhiteSpace(sql)) throw new ArgumentException("SQL statement is required.", nameof(sql));
   323
   324	            sql = sql.Trim();
   325	            if (sql.EndsWith(";"))
   326	                sql = sql.TrimEnd(';').TrimEnd();
   327
   328	            if (sql.IndexOf(';') >= 0)
   329	                throw new InvalidOperationException("Multiple SQL statements are not allowed.");
   330
   331	            var normalizedMode = mode?.Trim().ToLowerInvariant() ?? "select";
   332	            if (normalizedMode != "select" && normalizedMode != "nonquery" && normalizedMode != "scalar")
   333	                throw new ArgumentException("Invalid SQL execution mode. Allowed values are select, nonquery, scalar.", nameof(mode));
   334
   335	            var placeholderCount = 0;
   336	            foreach (var ch in sql)
   337	            {
   338	                if (ch == '?')
   339	                    placeholderCount++;
   340	            }
   341
   342	            if (parameters != null && placeholderCount != parameters.Count)
   343	                throw new ArgumentException($"SQL parameter count mismatch. Expected {placeholderCount}, got {parameters.Count}.", nameof(parameters));
   344
   345	            using var command = new OleDbCommand(sql, _oleDbConnection)
   346	            {
   347	                CommandType = CommandType.Text
   348	            };
   349
   350	            if (parameters != null)
   351	            {
   352	                foreach (var parameter in parameters)
   353	                {
   354	                    command.Parameters.Add(CreateOleDbParameter(parameter));
   355	                }
   356	            }
   357
   358	            if (normalizedMode == "scalar")
   359	            {
   360	                return command.ExecuteScalar();
   361	            }
   362
   363	            if (normalizedMode == "nonquery")
   364	            {
   365	                return command.ExecuteNonQuery();
   366	            }
   367
   368	            using var reader = command.ExecuteReader();
   369	            var rows = new List<Dictionary<string, object?>>();
   370	            while (reader.Read())
   371	            {
   372	                var row = new Dictionary<string, object?>();
   373	                for (int i = 0; i < reader.FieldCount; i++)
   374	                {
   375	                    var name = reader.GetName(i);
   376	                    var value = reader.IsDBNull(i) ? null : reader.GetValue(i);
   377	                    row[name] = value;
   378	                }
   379	                rows.Add(row);
   380	            }
   381	            return rows;
   382	        }

   384	        public List<Dictionary<string, object?>> ReadTableData(string objectName, int limit = 50, int offset = 0)
   385	        {
   386	            if (!IsConnected) throw new InvalidOperationException("Not connected to database");
   387	            if (string.IsNullOrWhiteSpace(objectName)) throw new ArgumentException("Object name is required.", nameof(objectName));
   388	            if (!IsValidObjectName(objectName)) throw new ArgumentException("Invalid object name.", nameof(objectName));
   389	            if (limit <= 0) limit = 50;
   390	            if (offset < 0) offset = 0;
   391
   392	            var sql = $"SELECT * FROM [{objectName}]";
   393	            using var command = new OleDbCommand(sql, _oleDbConnection)
   394	            {
   395	                CommandType = CommandType.Text
   396	            };
   397
   398	            using var reader = command.ExecuteReader();
   399	            var rows = new List<Dictionary<string, object?>>();
   400	            var skipped = 0;
   401	            while (reader.Read())
   402	            {
   403	                if (skipped < offset)
   404	                {
   405	                    skipped++;
   406	                    continue;
   407	                }
   408
   409	                var row = new Dictionary<string, object?>(StringComparer.OrdinalIgnoreCase);
   410	                for (var i = 0; i < reader.FieldCount; i++)
   411	                {
   412	                    var name = reader.GetName(i);
   413	                    var rawValue = reader.IsDBNull(i) ? null : reader.GetValue(i);
   414	                    row[name] = NormalizeValue(rawValue);
   415	                }
   416	                rows.Add(row);
   417
   418	                if (rows.Count >= limit)
   419	                    break;
   420	            }
   421
   422	            return rows;
   423	        }

   425	        public object RunMacroOrVBA(string name, List<object?>? arguments = null)
   426	        {
   427	            if (string.IsNullOrWhiteSpace(name))
   428	                throw new ArgumentException("Macro or function name is required.", nameof(name));
   429
   430	            var accessApp = EnsureAccessApplication();
   431	            if (accessApp == null)
   432	                throw new InvalidOperationException("Access application is not available. Please launch Access first.");
   433
   434	            try
   435	            {
   436	                var invocationArgs = new List<object?> { name };
   437	                if (arguments != null)
   438	                    invocationArgs.AddRange(arguments);
   439
   440	                var result = accessApp.GetType().InvokeMember("Run", BindingFlags.InvokeMethod, null, accessApp, invocationArgs.ToArray());
   441	                return NormalizeValue(result);
   442	            }
   443	            catch (System.Runtime.InteropServices.COMException comEx)
   444	            {
   445	                throw new InvalidOperationException($"Failed to run macro or VBA function '{name}': {comEx.Message}", comEx);
   446	            }
   447	            catch (TargetInvocationException tie) when (tie.InnerException != null)
   448	            {
   449	                throw new InvalidOperationException($"Failed to run macro or VBA function '{name}': {tie.InnerException.Message}", tie.InnerException);
   450	            }
   451	            catch (Exception ex)
   452	            {
   453	                throw new InvalidOperationException($"Failed to run macro or VBA function '{name}': {ex.Message}", ex);
   454	            }
   455	        }

   457	        public string GetFullSchemaMarkdown()
   458	        {
   459	            if (!IsConnected) throw new InvalidOperationException("Not connected to database");
   460
   461	            var tables = GetTables();
   462	            var relationships = GetRelationships();
   463	            var builder = new StringBuilder();
   464
   465	            builder.AppendLine("# Database schema");
   466	            builder.AppendLine();
   467	            builder.AppendLine("## Tables");
   468	            builder.AppendLine();
   469
   470	            foreach (var table in tables)
   471	            {
   472	                builder.AppendLine($"### {EscapeMarkdown(table.Name)} ({table.RecordCount} rows)");
   473	                builder.AppendLine();
   474	                builder.AppendLine("| Field | Type | Size | Required | AllowZeroLength |");
   475	                builder.AppendLine("|---|---|---|---|---|");
   476	                foreach (var field in table.Fields)
   477	                {
   478	                    builder.AppendLine($"| {EscapeMarkdown(field.Name)} | {EscapeMarkdown(field.Type)} | {field.Size} | {field.Required} | {field.AllowZeroLength} |");
   479	                }
   480	                builder.AppendLine();
   481	            }
   482
   483	            builder.AppendLine("## Relationships");
   484	            builder.AppendLine();
   485	            builder.AppendLine("| Name | Table | Foreign Table | Attributes |");
   486	            builder.AppendLine("|---|---|---|---|");
   487	            foreach (var relationship in relationships)
   488	            {
   489	                builder.AppendLine($"| {EscapeMarkdown(relationship.Name)} | {EscapeMarkdown(relationship.Table)} | {EscapeMarkdown(relationship.ForeignTable)} | {EscapeMarkdown(relationship.Attributes)} |");
   490	            }
   491
   492	            return builder.ToString().Trim();
   493	        }

   534	        private static object? NormalizeValue(object? value)
   535	        {
   536	            if (value == null || value == DBNull.Value)
   537	                return null;
   538
   539	            return value switch
   540	            {
   541	                byte[] bytes => Convert.ToBase64String(bytes),
   542	                DateTime dateTime => dateTime.ToString("o"),
   543	                Guid guid => guid.ToString(),
   544	                char character => character.ToString(),
   545	                string _ => value,
   546	                bool _ => value,
   547	                byte _ => value,
   548	                sbyte _ => value,
   549	                short _ => value,
   550	                ushort _ => value,
   551	                int _ => value,
   552	                uint _ => value,
   553	                long _ => value,
   554	                ulong _ => value,
   555	                float _ => value,
   556	                double _ => value,
   557	                decimal _ => value,
   558	                _ => value.ToString()
   559	            };
   560	        }

   562	        private static bool IsValidObjectName(string name)
   563	        {
   564	            if (string.IsNullOrWhiteSpace(name))
   565	                return false;
   566
   567	            if (name.IndexOfAny(new[] { '[', ']', ';', '\'', '"' }) >= 0)
   568	                return false;
   569
   570	            foreach (var character in name)
   571	            {
   572	                if (!(char.IsLetterOrDigit(character) || character == '_' || character == ' ' || character == '$' || character == '#' || character == '.'))
   573	                    return false;
   574	            }
   575
   576	            return true;
   577	        }

   579	        private static string EscapeMarkdown(string value)
   580	        {
   581	            if (string.IsNullOrEmpty(value))
   582	                return string.Empty;
   583
   584	            return value.Replace("|", "\\|").Replace("\r", " ").Replace("\n", " ");
   585	        }

   587	        private static string SanitizeIdentifier(string input)
   588	        {
   589	            if (string.IsNullOrWhiteSpace(input))
   590	                return "_";
   591
   592	            var builder = new StringBuilder();
   593	            foreach (var ch in input)
   594	            {
   595	                if (char.IsLetterOrDigit(ch) || ch == '_')
   596	                    builder.Append(ch);
   597	                else
   598	                    builder.Append('_');
   599	            }
   600
   601	            var result = builder.ToString();
   602	            if (result.Length == 0)
   603	                return "_";
   604
   605	            if (!char.IsLetter(result[0]) && result[0] != '_')
   606	                result = "_" + result;
   607
   608	            return result;
   609	        }

   611	        private static string SanitizeFileName(string input)
   612	        {
   613	            if (string.IsNullOrWhiteSpace(input))
   614	                return Guid.NewGuid().ToString("N");
   615
   616	            var builder = new StringBuilder();
   617	            foreach (var ch in input)
   618	            {
   619	                if (Path.GetInvalidFileNameChars().Contains(ch) || char.IsWhiteSpace(ch))
   620	                    builder.Append('_');
   621	                else
   622	                    builder.Append(ch);
   623	            }
   624
   625	            var name = builder.ToString();
   626	            return string.IsNullOrEmpty(name) ? Guid.NewGuid().ToString("N") : name;
   627	        }

   629	        private static string MapAccessTypeToCSharpType(string accessType)
   630	        {
   631	            if (string.IsNullOrWhiteSpace(accessType))
   632	                return "string";
   633
   634	            if (int.TryParse(accessType, out var typeCode))
   635	            {
   636	                return typeCode switch
   637	                {
   638	                    2 => "int",
   639	                    3 => "int",
   640	                    4 => "float",
   641	                    5 => "double",
   642	                    6 => "decimal",
   643	                    7 => "DateTime",
   644	                    10 => "DateTime",
   645	                    11 => "bool",
   646	                    17 => "byte",
   647	                    72 => "Guid",
   648	                    128 => "byte[]",
   649	                    130 => "string",
   650	                    201 => "string",
   651	                    203 => "string",
   652	                    204 => "byte[]",
   653	                    205 => "byte[]",
   654	                    _ => "string",
   655	                };
   656	            }
   657
   658	            var normalized = accessType.Trim().ToLowerInvariant();
   659	            if (normalized.Contains("char") || normalized.Contains("text") || normalized.Contains("memo") || normalized.Contains("string") || normalized.Contains("varchar"))
   660	                return "string";
   661	            if (normalized.Contains("date") || normalized.Contains("time"))
   662	                return "DateTime";
   663	            if (normalized.Contains("bool"))
   664	                return "bool";
   665	            if (normalized.Contains("currency") || normalized.Contains("decimal") || normalized.Contains("numeric"))
   666	                return "decimal";
   667	            if (normalized.Contains("double"))
   668	                return "double";
   669	            if (normalized.Contains("single") || normalized.Contains("float"))
   670	                return "float";
   671	            if (normalized.Contains("byte") || normalized.Contains("binary") || normalized.Contains("oleobject"))
   672	                return "byte[]";
   673	            if (normalized.Contains("guid"))
   674	                return "Guid";
   675
   676	            return "string";
   677	        }
